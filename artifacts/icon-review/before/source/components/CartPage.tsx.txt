"use client";

import { useRef } from "react";
import Link from "next/link";
import { ArrowLeft, ArrowRight, Minus, Plus, X } from "lucide-react";
import { formatPrice, variantById } from "@/data/products";
import { useCart } from "./CartProvider";
import { ProductArtwork } from "./ProductArtwork";

export function CartPage() {
  const { items, count, subtotalCents, ready, changeQuantity, remove } = useCart();
  const summary = useRef<HTMLDialogElement>(null);
  const checkoutTrigger = useRef<HTMLButtonElement>(null);
  if (!ready) return <main id="main-content" className="cart-page"><div className="page-intro"><p className="eyebrow">YOUR SELECTION / 00</p><h1>THE BAG<span className="accent-dot">.</span></h1></div><p aria-live="polite">Loading bag…</p></main>;
  return <main id="main-content" className="cart-page"><div className="page-intro"><p className="eyebrow">YOUR SELECTION / {count.toString().padStart(2, "0")}</p><h1>THE BAG<span className="accent-dot">.</span></h1></div>
    {!items.length ? <section className="empty-cart"><p>00 / NOTHING HELD BACK</p><h2>YOUR BAG IS<br />A BLANK CANVAS.</h2><Link className="black-button" href="/shop">EXPLORE THE COLLECTION <ArrowRight size={20} /></Link></section> : <div className="cart-layout"><div className="cart-lines">{items.map(item => { const match = variantById(item.variantId); if (!match) return null; const { product, variant } = match; return <article className="cart-line" key={item.variantId}><Link href={`/products/${product.slug}`} className="cart-art"><ProductArtwork product={product} /></Link><div className="cart-line-info"><div><p className="eyebrow">{product.code} / {product.category.toUpperCase()}</p><Link href={`/products/${product.slug}`} className="cart-title">{product.name}</Link><p>{variant.color} / {variant.size}</p></div><div className="cart-line-bottom"><div className="quantity-control"><button type="button" onClick={() => changeQuantity(item.variantId, item.quantity - 1)} aria-label={`Decrease quantity of ${product.name} size ${variant.size}`}><Minus size={15} /></button><span aria-label={`Quantity ${item.quantity}`}>{item.quantity}</span><button type="button" onClick={() => changeQuantity(item.variantId, item.quantity + 1)} disabled={item.quantity >= 99} aria-label={`Increase quantity of ${product.name} size ${variant.size}`}><Plus size={15} /></button></div><strong>{formatPrice(product.priceCents * item.quantity)}</strong></div></div><button type="button" className="remove-line" onClick={() => remove(item.variantId)} aria-label={`Remove ${product.name} size ${variant.size}`}><X size={18} /></button></article>; })}<Link className="continue-shopping" href="/shop"><ArrowLeft size={17} /> CONTINUE SHOPPING</Link></div><aside className="cart-summary"><p className="eyebrow">ORDER SUMMARY / DEMO</p><h2>YOUR<br />SELECTION.</h2><div><span>SUBTOTAL / {count} ITEMS</span><strong>{formatPrice(subtotalCents)}</strong></div><p>Shipping and taxes are not calculated for this concept project.</p><button ref={checkoutTrigger} type="button" className="black-button" onClick={() => summary.current?.showModal()}>CHECKOUT DEMO <ArrowRight size={20} /></button><small>No payment or order is processed.</small></aside></div>}
    <dialog ref={summary} className="demo-dialog" aria-labelledby="demo-heading" onClose={() => checkoutTrigger.current?.focus()} onClick={event => { if (event.target === summary.current) summary.current?.close(); }}><div className="demo-card"><button type="button" aria-label="Close demo summary" onClick={() => summary.current?.close()}><X /></button><p className="eyebrow">FORM / 27 / CONCEPT PROJECT</p><h2 id="demo-heading">A STUDY IN<br />COMMERCE.</h2><p>Your {count} selected {count === 1 ? "piece" : "pieces"} total {formatPrice(subtotalCents)}. This is a portfolio demo: no payment information is collected and no order is created.</p><button type="button" className="outline-button" onClick={() => summary.current?.close()}>RETURN TO BAG ↗</button></div></dialog>
  </main>;
}
